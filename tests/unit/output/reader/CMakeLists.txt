cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(Output2DReaderTests LANGUAGES C CXX)
    enable_testing()
    option(OPENSWMM_OUTPUT2D_TEST_HDF5 "Exercise the HDF5 reader backend" ON)
else()
    set(OPENSWMM_OUTPUT2D_TEST_HDF5 OFF)
    if(OPENSWMM_BUILD_2D AND OPENSWMM_2D_HDF5_OUTPUT)
        set(OPENSWMM_OUTPUT2D_TEST_HDF5 ON)
    endif()
endif()
find_package(GTest CONFIG REQUIRED)
get_filename_component(_reader_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
# Compile the reader in isolation: no solver/Qt runtime, and exactly one HDF5
# instance in this process. Exercise the public C ABI as a standalone consumer.
add_executable(test_output2d
    ../test_output2d.cpp
    "${_reader_root}/src/engine/output/Output2DReader.cpp"
    "${_reader_root}/src/engine/output/openswmm_output2d_impl.cpp")
target_compile_features(test_output2d PRIVATE cxx_std_20)
target_include_directories(test_output2d PRIVATE "${_reader_root}/include")
target_compile_definitions(test_output2d PRIVATE OPENSWMM_ENGINE_STATIC
    OUTPUT2D_TEST_DIRECTORY="${_reader_root}/tests/output/output2d_reader")
target_link_libraries(test_output2d PRIVATE GTest::gtest_main)
if(OPENSWMM_OUTPUT2D_TEST_HDF5)
    find_package(hdf5 CONFIG QUIET)
    if(TARGET hdf5::hdf5-static)
        target_link_libraries(test_output2d PRIVATE hdf5::hdf5-static)
    elseif(TARGET hdf5::hdf5-shared)
        target_link_libraries(test_output2d PRIVATE hdf5::hdf5-shared)
    else()
        find_package(HDF5 REQUIRED COMPONENTS C)
        target_link_libraries(test_output2d PRIVATE ${HDF5_C_LIBRARIES})
        target_include_directories(test_output2d PRIVATE ${HDF5_INCLUDE_DIRS})
    endif()
    target_compile_definitions(test_output2d PRIVATE OPENSWMM_HAS_2D_HDF5_OUTPUT=1)
endif()
add_test(NAME test_output2d COMMAND test_output2d)
set_tests_properties(test_output2d PROPERTIES LABELS "unit;output;2d")
